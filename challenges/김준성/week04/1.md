#include <stdio.h>
#include <string.h>

// 프로그램의 목적을 알리는 배너 출력
void print_banner() {
    printf("========================================\n");
    printf("    [Week 04] Caesar Cipher System      \n");
    printf("========================================\n");
}

// 사용자로부터 shift 값을 입력받아 반환
int get_shift() {
    int shift;
    printf("Enter shift value (1-25): ");
    scanf("%d", &shift);
    return shift;
}

// 문자열을 shift만큼 밀어서 암호화 (원본 문자열 직접 수정)
void encode(char *str, int shift) {
    for (int i = 0; str[i] != '\0'; i++) {
        char c = str[i];
        
        // 소문자 처리
        if (c >= 'a' && c <= 'z') {
            str[i] = (c - 'a' + shift) % 26 + 'a';
        }
        // 대문자 처리
        else if (c >= 'A' && c <= 'Z') {
            str[i] = (c - 'A' + shift) % 26 + 'A';
        }
    }
}

int main() {
    char message[100];
    int shift;

    print_banner();

    printf("Input Message: ");
    // 공백을 포함한 문자열 입력을 위해 fgets 사용
    fgets(message, sizeof(message), stdin);
    message[strcspn(message, "\n")] = 0; // 개행 문자 제거

    shift = get_shift();

    encode(message, shift);
    printf("Encoded Result: %s\n", message);

    // 다시 반대로 밀면 디코딩이 됩니다.
    encode(message, 26 - (shift % 26));
    printf("Decoded Result: %s\n", message);

    return 0;
}


함수를 쓰는이유: 코드의 중복을 없애 재사용성을 높이고, 복잡한 로직을 기능별로 분리하여 프로그램의 가독성과 유지보수성을 극대화하기 위해 사용합니다. 
유용한 상황: 특정 계산이나 출력이 반복적으로 필요할 때, 혹은 프로그램의 전체 흐름을 논리적 단계로 명확히 나누어 관리하고 싶을 때 매우 유용합니다.
